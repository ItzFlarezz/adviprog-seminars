type ProductCardProps = {
  image: string;
  category: string;
  name: string;
  description: string;
  price: number;
  onAddToCart: () => void;
};

function ProductCard({
  image,
  category,
  name,
  description,
  price,
  onAddToCart,
}: 

ProductCardProps) {
  return (
    <article className="w-full max-w-2xl overflow-hidden rounded-2xl bg-white shadow-lg md:flex">
      <img
        className="h-56 w-full object-cover md:h-auto md:w-2/5"
        src={image}
        alt={name}
      />

      <div className="p-6">
        <p className="text-xs font-semibold tracking-widest text-indigo-600">
          {category}
        </p>

        <h2 className="mt-2 text-2xl font-bold">
          {name}
        </h2>

        <p className="mt-3 text-sm leading-6 text-slate-600">
          {description}
        </p>

        <div className="mt-6 flex items-center justify-between">
          <span className="text-xl font-bold">
            ${price}
          </span>

          <button
            onClick={onAddToCart}
            className="
              rounded-xl
              bg-indigo-600
              px-4
              py-2
              text-sm
              font-semibold
              text-white
              transition
              hover:bg-indigo-700
              active:scale-95
              focus:outline-none
              focus:ring-2
              focus:ring-indigo-500
              focus:ring-offset-2
            "
          >
            Add to cart
          </button>
        </div>
      </div>
    </article>
  );
}

export default ProductCard;
